# Ficha de 7º Mar v2 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Completar o Criador de Fichas de 7º Mar — adicionar Feitiçaria, Estilos de Duelo, Histórias Heroicas, Sociedades Secretas e Toques Finais, transformar a ferramenta numa **ficha viva** (modo Jogo) com criação em abas, e modularizar o código.

**Architecture:** Uma única página (`setimo-mar.html`) com um toggle **Criação | Jogo**. A Criação é um assistente em abas; o Jogo é uma ficha viva editável. O `builder.js` da v1 é quebrado em módulos com responsabilidade única; `state.js` é a única fonte de verdade do estado, persistido em `localStorage` (`setimoMar.ficha.v2`) com migração automática do v1.

**Tech Stack:** HTML + CSS + JavaScript puro (ES2020), sem build, sem dependências externas. Testes em Node puro (sem framework).

## Global Constraints

- Sem build, sem bundler, sem dependências externas na página de 7º Mar (não carregar Bootstrap ali). — verbatim da spec §2.
- `state.js` é a única fonte de verdade do estado. — spec §4.3.
- Persistência em `localStorage`, chave `setimoMar.ficha.v2`, campo `schemaVersion: 2`; migração automática do `setimoMar.ficha.v1` sem apagar a chave v1. — spec §5.1.
- Dados de Feitiçaria/Duelo/Estilos/Manobras são **curados** e cada item marcado com `nota: "curado"` (ou comentário) para revisão. — spec §7.
- Tudo em português, presente de 1668 A.V. — convenção do projeto.
- Testes: `node tests/rules.test.js` (Node puro). Nenhuma alegação de "pronto" sem a saída dos testes + evidência de smoke no navegador. — spec §8.
- Trabalho apenas na branch `setimo-mar`. **Os commits são feitos pelo usuário** — cada passo "Commit" mostra o comando; o usuário o executa.
- Nomes de arquivo sem acento; tema/estilo consistentes com a v1 (`styles-setimo-mar.css`).

```js
// Carrega os módulos do builder sob stubs de browser e devolve o escopo global.
const fs = require("fs");
const path = require("path");
const vm = require("vm");

function makeStubs() {
  const store = {};
  const el = () => ({ innerHTML: "", style: {}, value: "", textContent: "",
    classList: { add(){}, remove(){}, toggle(){} }, appendChild(){}, remove(){},
    click(){}, setAttribute(){}, querySelectorAll: () => [], scrollIntoView(){} });
  const sandbox = {};
  sandbox.window = sandbox;
  sandbox.localStorage = {
    getItem: k => (k in store ? store[k] : null),
    setItem: (k, v) => { store[k] = String(v); },
    removeItem: k => { delete store[k]; },
  };
  sandbox.document = {
    addEventListener(){}, getElementById: () => null,
    querySelector: () => null, querySelectorAll: () => [],
    createElement: () => el(), body: el(),
  };
  sandbox.confirm = () => true;
  sandbox.alert = () => {};
  sandbox.URL = { createObjectURL: () => "blob:x", revokeObjectURL(){} };
  sandbox.Blob = class {};
  sandbox.FileReader = class {};
  sandbox.console = console;
  sandbox._store = store;
  return sandbox;
}

function loadSM(files, presetStore) {
  const sandbox = makeStubs();
  if (presetStore) Object.assign(sandbox._store, presetStore);
  vm.createContext(sandbox);
  const root = path.join(__dirname, "..");
  for (const f of files) {
    const code = fs.readFileSync(path.join(root, f), "utf8");
    vm.runInContext(code, sandbox, { filename: f });
  }
  return sandbox;
}

// mini-assert
let pass = 0, fail = 0;
function eq(name, a, b) {
  const A = JSON.stringify(a), B = JSON.stringify(b);
  if (A === B) pass++;
  else { fail++; console.log(`FAIL ${name}: esperado ${B}, obteve ${A}`); }
}
function ok(name, cond) { eq(name, !!cond, true); }
function report() {
  console.log(`\n=== ${pass} passaram, ${fail} falharam ===`);
  process.exit(fail ? 1 : 0);
}

module.exports = { loadSM, eq, ok, report };
```

```js
function smSave() {
  try { localStorage.setItem(SM_STORAGE_KEY, JSON.stringify(SM)); } catch(e){}

function smMergeDefaults(data) {
  const base = smDefaultState();
  const out = Object.assign(base, data);
  // garante sub-objetos e perícias
  out.play = Object.assign(base.play, data.play || {});
  out.sorcery = Object.assign(base.sorcery, data.sorcery || {});
  out.dueling = Object.assign(base.dueling, data.dueling || {});
  out.society = Object.assign(base.society, data.society || {});

  SM_SKILLS.forEach(s => { if (out.skills[s.key] == null) out.skills[s.key] = 0; });
  out.schemaVersion = 2;
  return out;
}
function smLoad() {
  try {

    if (rawV2) { SM = smMergeDefaults(JSON.parse(rawV2)); return true; }
    const rawV1 = localStorage.getItem(SM_STORAGE_KEY_V1);
    if (rawV1) { SM = smMergeDefaults(JSON.parse(rawV1)); smSave(); return true; } // migra e persiste em v2
    return false;
  } catch(e) { return false; }
}
```

```js
// ══ 7º MAR — App: modos e abas ══
const SM_TABS = [
  { id:"conceito",  label:"1 · Conceito & Nação" },
  { id:"atributos", label:"2 · Atributos" },
  { id:"formacoes", label:"3 · Formações" },
  { id:"pericias",  label:"4 · Perícias" },
  { id:"vantagens", label:"5 · Vantagens" },
  { id:"arcanos",   label:"6 · Arcanos" },
  { id:"feiticaria",label:"7 · Feitiçaria", cond:() => smTemFeiticaria() },
  { id:"duelo",     label:"8 · Duelo",      cond:() => smTemAcademiaDuelista() },
  { id:"historias", label:"9 · Histórias" },
  { id:"sociedade", label:"10 · Sociedade" },
  { id:"toques",    label:"11 · Toques Finais" },
  { id:"revisao",   label:"12 · Revisão" },
];
let SM_TAB_INDEX = 0;

function smVisibleTabs() { return SM_TABS.filter(t => !t.cond || t.cond()); }

function smTabComplete(id) {
  switch (id) {
    case "conceito":  return !!SM.concept.name && !!SM.nation && !!SM.nationTrait;
    case "atributos": return smTraitFreeLeft() === 0;
    case "formacoes": return SM.backgrounds.length === SM_CREATION.backgroundsToPick;
    case "pericias":  return smSkillPointsLeft() === 0;
    case "vantagens": return smAdvantagePointsLeft() >= 0; // sempre ok (opcional gastar tudo)
    case "arcanos":   return !!SM.arcana.virtude && !!SM.arcana.hubris;
    case "feiticaria":return !smTemFeiticaria() || !!SM.sorcery.school;
    case "duelo":     return !smTemAcademiaDuelista() || !!SM.dueling.style;
    case "historias": return true;   // opcional na criação
    case "sociedade": return true;   // opcional
    case "toques":    return true;   // opcional
    case "revisao":   return true;
    default: return true;
  }
}

function smSetMode(mode) {           // "criar" | "jogo"
  SM._mode = mode;
  document.getElementById("sm-mode-criar").classList.toggle("sm-active", mode==="criar");
  document.getElementById("sm-mode-jogo").classList.toggle("sm-active", mode==="jogo");
  document.getElementById("sm-create-panel").style.display = mode==="criar" ? "block" : "none";
  document.getElementById("sm-play").style.display = mode==="jogo" ? "block" : "none";
  if (mode==="jogo") renderPlay();
}
function smGoTab(index) {
  const tabs = smVisibleTabs();
  SM_TAB_INDEX = Math.max(0, Math.min(index, tabs.length-1));
  const active = tabs[SM_TAB_INDEX].id;
  document.querySelectorAll(".sm-tab-panel").forEach(p =>
    p.style.display = (p.dataset.tab === active) ? "block" : "none");
  renderTabBar();
  window.scrollTo({ top: 0, behavior: "smooth" });
}
function smNextTab() { smGoTab(SM_TAB_INDEX + 1); }
function smPrevTab() { smGoTab(SM_TAB_INDEX - 1); }

function renderTabBar() {
  const bar = document.getElementById("sm-tab-bar");
  if (!bar) return;
  const tabs = smVisibleTabs();
  bar.innerHTML = tabs.map((t, i) => {
    const done = smTabComplete(t.id) ? "✓" : "";
    const on = i === SM_TAB_INDEX ? " sm-tab-active" : "";
    return `<button class="sm-tab${on}" onclick="smGoTab(${i})">${t.label} <span class="sm-tab-check">${done}</span></button>`;
  }).join("");
}

function smInit() {
  smLoad();
  renderAll();          // create.js
  renderTabBar();
  smSetMode(SM._mode || "criar");
  smGoTab(0);
}
document.addEventListener("DOMContentLoaded", smInit);

Object.assign(window, { smSetMode, smGoTab, smNextTab, smPrevTab, smTabComplete, SM_TABS });
```
